import os import speech_recognition as sr from gtts import gTTS import playsound # Hoặc dùng thư viện âm thanh khác # Giả sử sử dụng thư viện Google GenAI SDK mới nhất from google import genai # Khởi tạo client AI (đảm bảo bạn đã thiết lập biến môi trường GEMINI_API_KEY) client = genai.Client() def speak(text): """Chuyển văn bản thành giọng nói và phát ra loa""" print(f"🤖 AI: {text}") tts = gTTS(text=text, lang="vi", slow=False) audio_file = "response.mp3" tts.save(audio_file) playsound.playsound(audio_file) os.remove(audio_file) def listen_microphone(): """Thu âm từ micro của người dùng và chuyển thành văn bản""" r = sr.Recognizer() with sr.Microphone() as source: print("🎙️ Đang lắng nghe... Hãy nói gì đó:") r.adjust_for_ambient_noise(source) audio = r.listen(source) try: # Sử dụng nhận diện tiếng Việt text = r.recognize_google(audio, language="vi-VN") print(f"👤 Bạn: {text}") return text except sr.UnknownValueError: print("⚠️ Không thể nhận dạng được giọng nói.") return None except sr.RequestError as e: print(f"⚠️ Lỗi kết nối dịch vụ nhận dạng giọng nói: {e}") return None def main(): # Chức năng chào hỏi speak("Xin chào! Tôi là trợ lý ảo AI của bạn. Tôi có thể giúp gì cho bạn?") while True: # Tiếp nhận thông tin từ người dùng qua Mic user_input = listen_microphone() if user_input: if "thoát" in user_input.lower() or "tạm biệt" in user_input.lower(): speak("Tạm biệt bạn! Hẹn gặp lại.") break # Gửi thông tin đến mô hình AI xử lý try: response = client.models.generate_content( model="gemini-2.5-flash", contents=user_input, ) ai_reply = response.text # Phát giọng đọc AI speak(ai_reply) except Exception as e: speak("Xin lỗi, tôi đang gặp sự cố khi xử lý yêu cầu.") print(f"Lỗi: {e}") if __name__ == "__main__": main()
